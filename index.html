        function processAILogic(weather, occasion) {

            // "AI" matching algorithm: Filter by occasion AND weather

            const matchingTops = closetData.filter(i => i.type === 'top' && (i.category === occasion || i.category === 'casual') && i.weather.includes(weather));

            const matchingBottoms = closetData.filter(i => i.type === 'bottom' && (i.category === occasion || i.category === 'casual') && i.weather.includes(weather));

            const matchingShoes = closetData.filter(i => i.type === 'shoes' && (i.category === occasion || i.category === 'casual') && i.weather.includes(weather));

            const matchingAccessories = closetData.filter(i => i.type === 'accessory' && (i.category === occasion || i.category === 'casual'));

            

            // Fallbacks in case filter is too strict (preventing empty arrays)

            const top = matchingTops.length > 0 ? matchingTops[Math.floor(Math.random() * matchingTops.length)] : closetData.find(i => i.type === 'top');

            const bottom = matchingBottoms.length > 0 ? matchingBottoms[Math.floor(Math.random() * matchingBottoms.length)] : closetData.find(i => i.type === 'bottom');

            const shoes = matchingShoes.length > 0 ? matchingShoes[Math.floor(Math.random() * matchingShoes.length)] : closetData.find(i => i.type === 'shoes');

            const accessory = matchingAccessories.length > 0 ? matchingAccessories[Math.floor(Math.random() * matchingAccessories.length)] : closetData.find(i => i.type === 'accessory');


            // Store current outfit globally so it can be saved if requested

            currentOutfitData = {

                id: Date.now(),

                bottom: bottom,

                shoes: shoes,

                occasion: occasion

            // Render Output

            renderOutfitItem('outfit-top', top);

        function handleAddItem(event) {

                showToast("Please select at least one weather condition.", true);

            if (!uploadedImageBase64) {

                image: uploadedImageBase64,

                emoji: '✨' // Fallback

            };


            // Add to database

            closetData.push(newItem);

            

            // Save to LocalStorage

            saveClosetToStorage();


            // Close modal and refresh grid

            closeAddModal();

            

            // Render the grid based on currently active filter

            const activeFilterBtn = document.querySelector('.filter-btn.active');

            const currentFilter = activeFilterBtn ? activeFilterBtn.dataset.filter : 'all';

            renderGrid(currentFilter);

            

            showToast("Item successfully added to your closet!");

        }


        // --- Routing & Toast Logic ---

        

        // Displays a temporary success or error message

        function showToast(message, isError = false) {

            const toast = document.getElementById('toast');

            const icon = document.getElementById('toast-icon');

            document.getElementById('toast-message').innerText = message;

            

            if (isError) {

                icon.className = 'fa-solid fa-circle-exclamation text-red-400 mr-3 text-lg';

            } else {

                icon.className = 'fa-solid fa-circle-check text-green-400 mr-3 text-lg';

            }

            

            toast.classList.remove('translate-y-24', 'opacity-0');

            setTimeout(() => {

                toast.classList.add('translate-y-24', 'opacity-0');

            }, 3000);

        }


        // Handles switching between sidebar tabs

        function switchTab(tabId, event) {

            if (event) event.preventDefault();

            

            // 1. Update Navigation UI active states

            document.querySelectorAll('.nav-item').forEach(el => {

                el.classList.remove('text-brand-500', 'bg-brand-50');

                el.classList.add('text-gray-500');

            });

            const activeNav = document.getElementById('nav-' + tabId);

            if (activeNav) {

                activeNav.classList.remove('text-gray-500');

                activeNav.classList.add('text-brand-500', 'bg-brand-50');

            }


            // 2. Hide all views, show targeted view

            document.querySelectorAll('.view-section').forEach(el => el.classList.add('hidden'));

            document.getElementById('view-' + tabId).classList.remove('hidden');


            // 3. Update Header Title & trigger specific renders

            const pageTitle = document.getElementById('page-title');

            const actionBtn = document.getElementById('header-action-btn');

            

            if (tabId === 'closet') {

                pageTitle.innerText = 'My Closet';

                actionBtn.classList.remove('hidden'); // Show Add Item button

                renderGrid('all');

            } else if (tabId === 'lookbook') {

                pageTitle.innerText = 'Lookbook';

                actionBtn.classList.add('hidden'); // Hide Add Item button

                renderLookbook();

            } else if (tabId === 'analytics') {

                pageTitle.innerText = 'Analytics';

                actionBtn.classList.add('hidden'); // Hide Add Item button

                renderAnalytics();

            }

        }


        // --- Lookbook Logic ---

        function saveToLookbook() {

            if (currentOutfitData) {

                savedOutfits.push(currentOutfitData);

                saveLookbookToStorage();

                closeModal();

                showToast("Outfit saved to Lookbook!");

            }

        }


        function renderLookbook() {

            const grid = document.getElementById('lookbook-grid');

            const emptyState = document.getElementById('lookbook-empty');

            

            grid.innerHTML = '';

            

            if (savedOutfits.length === 0) {

                emptyState.classList.remove('hidden');

                grid.classList.add('hidden');

                return;

            }


            emptyState.classList.add('hidden');

            grid.classList.remove('hidden');


            // Reverse array to show newest saved outfits first

            [...savedOutfits].reverse().forEach(outfit => {

                const card = document.createElement('div');

                card.className = 'bg-white rounded-2xl shadow-sm border border-gray-100 overflow-hidden group hover:shadow-md transition-shadow';

                

                // Helper to render image or emoji fallback for lookbook cards

                const getVisual = (item) => item.image 

                    ? `<img src="${item.image}" class="w-full h-full object-cover" alt="${item.name}">`

                    : `<div class="w-full h-full flex items-center justify-center bg-gray-50 text-4xl">${item.emoji}</div>`;


                card.innerHTML = `

                    <div class="p-4 border-b border-gray-50 flex justify-between items-center bg-gray-50/50">

                        <span class="text-xs font-semibold uppercase tracking-wider text-brand-500 bg-brand-50 px-2 py-1 rounded-md">

                            ${outfit.occasion} / ${outfit.weather}

                        </span>

                        <button onclick="removeOutfit(${outfit.id})" class="text-gray-400 hover:text-red-500 transition" title="Delete outfit">

                            <i class="fa-solid fa-trash-can"></i>

                        </button>

                    </div>

                    <!-- Visual layout of the outfit -->

                    <div class="flex h-48">

                        <div class="w-1/2 border-r border-gray-100 p-2 flex flex-col space-y-2 bg-white">

                            <div class="h-1/2 rounded-lg overflow-hidden border border-gray-100 relative shadow-inner">

                                ${getVisual(outfit.top)}

                            </div>

                            <div class="h-1/2 rounded-lg overflow-hidden border border-gray-100 relative shadow-inner">

                                ${getVisual(outfit.bottom)}

                            </div>

                        </div>

                        <div class="w-1/2 p-2 bg-white flex flex-col space-y-2">

                            <div class="h-1/2 w-full rounded-lg overflow-hidden border border-gray-100 relative shadow-inner">

                                ${getVisual(outfit.shoes)}

                            </div>

                            <div class="h-1/2 w-full rounded-lg overflow-hidden border border-gray-100 relative shadow-inner">

                                ${getVisual(outfit.accessory || {emoji: '✨', name: 'Accessory'})}

                            </div>

                        </div>

                    </div>

                `;

                grid.appendChild(card);

            });

        }


        function removeOutfit(id) {

            savedOutfits = savedOutfits.filter(o => o.id !== id);

            saveLookbookToStorage();

            renderLookbook();

            showToast("Outfit removed");

        }


        // --- Analytics Logic ---

        function renderAnalytics() {

            // 1. Calculate basic statistics

            const total = closetData.length;

            document.getElementById('stat-total').innerText = total;


            const typeCounts = { top: 0, bottom: 0, shoes: 0, outerwear: 0, accessory: 0 };

            const weatherCounts = {};

            const occasionCounts = {};


            // Parse database

            closetData.forEach(item => {

                if (typeCounts[item.type] !== undefined) typeCounts[item.type]++;

                occasionCounts[item.category] = (occasionCounts[item.category] || 0) + 1;

                item.weather.forEach(w => { weatherCounts[w] = (weatherCounts[w] || 0) + 1; });

            });


            // Determine top metrics

            const topType = Object.keys(typeCounts).reduce((a, b) => typeCounts[a] > typeCounts[b] ? a : b);

            const topOccasion = Object.keys(occasionCounts).reduce((a, b) => occasionCounts[a] > occasionCounts[b] ? a : b);

            const topWeather = Object.keys(weatherCounts).reduce((a, b) => weatherCounts[a] > weatherCounts[b] ? a : b);


            document.getElementById('stat-category').innerText = topType;

            document.getElementById('stat-occasion').innerText = topOccasion;

            document.getElementById('stat-weather').innerText = topWeather;


            // 2. Render Progress Bars for Closet Composition

            const compositionDiv = document.getElementById('analytics-composition');

            compositionDiv.innerHTML = ''; // Clear previous

            

            // Brand matching colors for the progress bars

            const colors = { top: 'bg-brand-500', bottom: 'bg-brand-700', shoes: 'bg-brand-900', outerwear: 'bg-gray-400', accessory: 'bg-yellow-500' };


            Object.keys(typeCounts).forEach(type => {

                const count = typeCounts[type];

                const percentage = Math.round((count / total) * 100) || 0;

                

                compositionDiv.innerHTML += `

                    <div>

                        <div class="flex justify-between text-sm font-medium mb-1 text-gray-700">

                            <span class="capitalize">${type}</span>

                            <span>${count} items (${percentage}%)</span>

                        </div>

                        <div class="w-full bg-gray-100 rounded-full h-2.5 overflow-hidden">

                            <div class="${colors[type]} h-2.5 rounded-full fade-in" style="width: ${percentage}%"></div>

                        </div>

                    </div>

                `;

            });

        }


        // Initialize App

        window.onload = () => {

            renderGrid('all');

        };

    </script>
